# The MS-FSCC cross-validation, taken once and kept as values.
#
# The design considered `smb-fscc` as a dev-dependency test oracle for its
# independently validated MS-FSCC struct layouts and **rejected it**: pinning
# does not stop a bot raising pull requests, and every `0.x` minor would be held
# for a human — precisely the cost that rejected the smb-rs family as a runtime
# dependency, re-entering through the development door. What it prescribes
# instead is this file: the cross-validation run once during implementation,
# its results committed as plain expected values beside the fixture bytes.
#
# **How it was run.** `smb-fscc 0.11.2` was taken into a scratch copy of this
# crate outside the repository, and the crate's own decoders were run against
# the same bytes as its `FileBothDirectoryInformation`,
# `FileBasicInformation`, `FileStandardInformation`, `FileFsSizeInformation`
# and `FileEndOfFileInformation`, field by field. Every field below agreed.
# There is no `Cargo.toml` entry for it here and there is not meant to be one:
# the value was taken once and the subscription dropped.
#
# **The bytes are the committed corpus's**, not synthesised, except where a
# line says otherwise. Three FIND replies were cross-validated in full — 113
# entries between them, from two server implementations — and only the first
# three entries of each are pinned here, the run having compared them all.
#
# **It is a partial oracle, and the limits below are measured rather than
# assumed.** Where it cannot speak to something this file says so instead of
# forcing an agreement.
#
# Format: one `name = value` per line. `#` opens a comment.


# ---------------------------------------------------------------------------
# What the crate could not validate. Each was run and each is the run's result.
# ---------------------------------------------------------------------------

# 1. The chained-list container. `ChainedItemList<T>` reads until a zero
#    `NextEntryOffset`, which SMB2 guarantees; an SMB1 FIND reply need not carry
#    one, and neither Samba corpus does — the last entry's offset runs to
#    exactly the end of the data block. So the container does not merely need
#    special-casing, it fails outright with "failed to fill whole buffer", on
#    the six-entry root listing and on the hundred-entry fragmented one alike.
#    The cross-validation therefore read each entry on its own, at the offsets
#    this crate's walker produced. What bounds this crate's walk is the data
#    block's length as well as the terminator, and that is why.
fscc_chained_list_decodes_an_smb1_find_reply = false
samba_root_chain_is_zero_terminated = false
samba_root_chain_last_offset = 664
samba_root_chain_data_length = 664

# 2. The information-class numbers. Its enum carries SMB2's, and every level
#    this port sends is SMB1's, so no class number could be compared.
fscc_both_directory_class = 0x0003
smb1_both_directory_class = 0x0104
fscc_basic_class = 0x0004
smb1_query_basic_class = 0x0101
fscc_standard_class = 0x0005
smb1_query_standard_class = 0x0102

# 3. The `EaSize` field of a directory entry. MS-FSCC gives that offset to
#    `ReparseTag` when `FILE_ATTRIBUTE_REPARSE_POINT` is set and to `EaSize`
#    otherwise, and `smb-fscc` switches on the attribute. [MS-CIFS] 2.2.8.1.7
#    defines `SMB_FIND_FILE_BOTH_DIRECTORY_INFO` with `EaSize` and no such
#    variant, so this port reads `EaSize` unconditionally and the two disagree
#    on a reparse-point entry by construction. Measured on an entry from the
#    corpus with the attribute set by hand: the port reports `EaSize`, the
#    crate reports a reparse tag and no `EaSize` at all.
fscc_reparse_point_entry_reports_ea_size = false
port_reparse_point_entry_ea_size = 0

# 4. The 8.3 short name. Its `FixedWideString<12>` reads all twelve units and
#    ignores `ShortNameLength`, so it cannot validate the trim. What it did
#    validate is that this crate's trimmed name is what remains of those twelve
#    units once the padding is removed — which held on all 113 entries, every
#    one of which carries an empty short name.
fscc_short_name_honours_the_length_field = false
corpus_short_names_are_all_empty = true

# 5. `SMB_INFO_ALLOCATION` (`0x0001`), the legacy level `Tree::statistics`
#    falls back to. It is SMB1-only and MS-FSCC defines no equivalent, so the
#    crate has nothing to compare against and this level is validated by the
#    fixture corpus and the live acceptance run alone.
fscc_has_an_smb_info_allocation_equivalent = false
