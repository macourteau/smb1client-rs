# The differential run of `session.rs` / `tree.rs` / `resource/` against the Go
# reference library, pinned at `b948f59`.
#
# Build order step 4 requires those three modules to be differentially tested
# against that library on the Samba container. **That run is development-time
# and never a CI job** — nothing in CI checks out the Go repository — so its
# comparisons are committed here as expected values and these are what CI
# asserts, the same arrangement `auth/` uses at step 3.
#
# **How the run was taken.** Both clients drove the seeded acceptance container
# at `127.0.0.1:10445` (`smbtest`/`smbtest`, share `testshare`) through the same
# operations in the same order — opening, reading, writing, listing, metadata,
# rename, delete, directory creation and the filesystem-statistics call — each
# through its own frame-splitting TCP relay, so every field below was read off
# the wire rather than off either library's own report. The reference was
# extracted at the pin with `git archive` and built in isolation: the checkout
# it came from sits on a branch 26 commits ahead, and those commits include
# fixes that adopt this port's own decisions, so a run against its working tree
# would disagree for reasons that are not the port's.
#
# **What the container negotiated**, which every derived number below rests on:
# `MaxBufferSize = 16,644`, `MaxMpxCount = 50`,
# `Capabilities = 0x8080F3FD` (both large-I/O bits set), Samba 4.23.8.
#
# **A disagreement here is read three ways and they are not the same rule.**
#
#  1. Where the design lists the reference as known-wrong, the port is right and
#     the entry records the correct value with the reference's beside it. Those
#     entries are prefixed `reference_` and each names the list item it belongs
#     to.
#  2. Where the reference disagrees with *itself* neither answer is the
#     reference and the run cannot adjudicate. The chunk-size block below is
#     that case, and it is the instance build order step 4 says to expect. What
#     is pinned there is the port's answer, because the port has one.
#  3. Anything else is a finding. Two are recorded below as `finding_` entries;
#     they are not asserted as reference behaviour and not reproduced.
#
# Format: one `name = value` per line. `#` opens a comment. Numbers are decimal
# unless they carry an `0x` prefix.

# ---------------------------------------------------------------------------
# Chunk size — the second reading, and the one build order step 4 names.
#
# **Neither answer is the reference here.** The reference's chunk size turns on
# which of its own four paths a transfer takes, and those paths do not agree
# with each other on the connection the run measured:
#
#   - a whole-file read under 128 KiB takes `readSequential`, which clamps to
#     `MaxBufferSize - 1024`, so 16,644 - 1,024 = 15,620;
#   - a whole-file read at or above 128 KiB takes `readPipelined`, which does
#     not consult the negotiated buffer at all and asks for 130,048;
#   - a `ReadAt` of any size takes `readAtChunk`, which does not consult it
#     either and asked for the whole 102,400 of a 100 KiB span;
#   - a write under 256 KiB takes `writeSequential`, clamped to that same
#     15,620, and one at or above it takes `writePipelined`'s flat 130,048.
#
# So one connection, one negotiated buffer, three different answers — and the
# design's own account of this had two, the unclamped `readAtChunk` path being
# a third the run found. The port sizes chunks by one rule whatever the
# transfer, so it necessarily differs from the reference on the sequential side
# of both thresholds. That disagreement is the reference's inconsistency and
# what is pinned is the port's answer.
#
# The port's rule has three cases and the run exercised the first; the other two
# are named here because they are what a wrong implementation would reach for.
# ---------------------------------------------------------------------------

container_negotiated_max_buffer_size = 16644
protocol_overhead_reservation = 1024

# The whole point of the entry above: 15,620 is where the port arrives only when

# ---------------------------------------------------------------------------
# The fill loop, visible on the wire.
#
# A 32-byte span asked for 8 bytes before end of file: the server answers 8, and
# the port re-issues the 24 it did not return rather than reading the short
# answer as the end. The reference stops at the first short chunk and reports
# `io.EOF` — known-wrong item 8 — which against this container happens to be the
# right answer and against Windows, which serves `min(asked, 65536)` on every
# read, loses the rest of the file silently.
#
# `read_exact_at` cannot fill the span either way, so the two libraries differ
# in what they return as well as in what they send; that difference is the
# fill-or-error contract the design chose and not a defect on either side.
# ---------------------------------------------------------------------------

port_read_chunks_short_far_end = 32,24
reference_read_chunks_short_far_end = 32

# `SESSION_SETUP_ANDX`, claiming three capabilities it does not implement:

# Finding 2. Deleting a non-empty directory reports success and deletes
# nothing, on both libraries. Samba answers the `NT_CREATE_ANDX` carrying
